# cplt default domain blocklist
#
# These domains are commonly used as exfiltration endpoints, C2 infrastructure,
# and reverse shell tunnels in real-world supply chain attacks (2025-2026).
#
# Sources:
#   - LOTS Project (Living Off Trusted Sites): https://lots-project.com/
#   - Socket: 60 malicious npm packages (2025)
#   - Mend.io: Shai-Hulud npm worm analysis (2025)
#   - Wiz: Shai-Hulud 2.0 — 25K+ repos exposed
#   - ReversingLabs: npm reverse shell malware
#   - Oligo: npm supply chain risks with AI agents
#
# Usage:
#   cplt --with-proxy --blocked-domains /path/to/blocked-domains.txt
#
# Or set in ~/.config/cplt/config.toml:
#   [proxy]
#   blocked_domains = "~/.config/cplt/blocked-domains.txt"
#
# This file is re-read on every proxy request — edit it live without restarting.

# ── Webhook and request capture services ──────────────────────────
# Primary exfiltration channel in observed npm/PyPI attacks.
# Attackers POST stolen credentials to disposable endpoints.
webhook.site
pipedream.com
requestbin.com
requestbin.net
hookbin.com
beeceptor.com
requestcatcher.com

# Used to stage stolen credentials for later retrieval.

rentry.co

# ── Messaging platforms used as C2 ───────────────────────────────
# Discord webhooks are the #1 observed C2 channel in npm attacks.
# Telegram bot API is write-only and hard to detect.
#
# NOTE: These are aggressive blocks. Uncomment ONLY if your workflow
# does not require these services from within the sandbox.
#
# discord.com
# api.telegram.org

# ── Generic cloud platforms used for C2 hosting ──────────────────
# Attackers host C2 on free-tier cloud to evade IP blocklists.
# These are very broad — uncomment only if you want strict lockdown.
#
# workers.dev
# herokuapp.com

# ── Developer tooling telemetry ──────────────────────────────────
# Build tools, framework CLIs, and language toolchains that send default-on
# usage analytics. These are not malicious, but represent unnecessary data
# collection from inside the sandbox. Tools continue to work normally.
# See also: HARDENING_ENV_VARS in sandbox_policy.rs (env-level opt-outs).

# HashiCorp tools (terraform, vault, packer, nomad) — version-check + usage pings
checkpoint.hashicorp.com

# Next.js — build telemetry (see also NEXT_TELEMETRY_DISABLED env var)
telemetry.nextjs.org

# VS Code extensions — Microsoft 1DS / OneCollector SDK used by all MS-authored
# extensions (Copilot, ESLint, GitLens variants, etc.) and VS Code itself
mobile.events.data.microsoft.com
dc.services.visualstudio.com

# Go toolchain — telemetry upload target (default is local-only, but block the
# upload endpoint to prevent accidental activation via `go telemetry on`)
telemetry.go.dev

# ── AI agent telemetry / analytics ───────────────────────────────
# AI coding agents send usage analytics and crash reports to third-party
# services. These are not malicious, but they represent unintended data
# exfiltration from inside the sandbox (code context, prompts, usage patterns).
#
# Blocked by default for privacy. Agents continue to function normally —
# these endpoints are non-essential telemetry.
posthog.com
